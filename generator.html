  <script>

    // —— 词库（已由粥粥提供） —— //

    const leftWords = [

      "打工","野人","熊老板","地图","大招","武器","环境","大鲑鱼","小鲑鱼",

      "垃圾桶","陀螺仪","视角","走位","这次身法","默契度","游泳圈","三角雷",

      "烤架鱼","光蝇","网络","加速器","任天堂"

    ];


    const rightWords = [

      "在干嘛","没吃饭吗","太烂了","实在不行","不合适","好差劲","不理解","有问题",

      "太糟了","全否定","没什么好说的","我讨厌你","特较真儿","一直在挑战我","好雷人",

      "没有被邀请","没有出现的义务","不对劲","搞啥呢","咋这样","我不同意","真是不懂了",

      "太安卓了","帮倒忙呢","没发力啊","真下饭","不中啊"

    ];


    // DOM

    const leftDisplay = document.getElementById('leftDisplay');

    const rightDisplay = document.getElementById('rightDisplay');

    const generateBtn = document.getElementById('generateBtn');

    const surpriseBtn = document.getElementById('surpriseBtn');

    const logArea = document.getElementById('log');

    const copyBtn = document.getElementById('copyBtn');

    const clearBtn = document.getElementById('clearBtn');


    // 记录数组，保证格式化输出

    const history = [];


    // 随机工具

    function randItem(arr){

      return arr[Math.floor(Math.random()*arr.length)];

    }


    // 生成一次（并追加到记录）

    function generateOnce({appendLog=true} = {}){

      const l = randItem(leftWords);

      const r = randItem(rightWords);


      leftDisplay.textContent = l;

      rightDisplay.textContent = r;


      if(appendLog){

        history.push(l + r);

        logArea.value = history.join('，'); // 使用中文逗号，保持示例一致

      }

    }


    // 事件绑定

    generateBtn.addEventListener('click', () => generateOnce({appendLog:true}));

    surpriseBtn.addEventListener('click', () => generateOnce({appendLog:true}));


    // 复制功能（使用 Clipboard API）

    copyBtn.addEventListener('click', async () => {

      const text = logArea.value;

      if(!text){

        copyBtn.textContent = '没有内容可复制';

        setTimeout(()=> copyBtn.textContent = '一键复制', 1200);

        return;

      }

      try {

        await navigator.clipboard.writeText(text);

        copyBtn.textContent = '已复制 ✓';

        setTimeout(()=> copyBtn.textContent = '一键复制', 1200);

      } catch (e) {

        // 退回到老式方法（兼容）

        logArea.select();

        try {

          document.execCommand('copy');

          copyBtn.textContent = '已复制 ✓';

          setTimeout(()=> copyBtn.textContent = '一键复制', 1200);

        } catch (err) {

          copyBtn.textContent = '复制失败';

          setTimeout(()=> copyBtn.textContent = '一键复制', 1200);

        }

      }

    });


    // 清空记录（但保留当前显示）

    clearBtn.addEventListener('click', () => {

      history.length = 0;

      logArea.value = '';

    });


    // 快速键：按空格生成（避免在输入框时触发）

    document.addEventListener('keydown', (e) => {

      if(e.code === 'Space' && document.activeElement.tagName !== 'TEXTAREA' && document.activeElement.tagName !== 'INPUT'){

        e.preventDefault();

        generateOnce({appendLog:true});

      }

    });


    // 初始化：先随机一次但不追加（页面打开时给个预览）

    generateOnce({appendLog:false});

  </script>
